\section*{The Gaussian integral}
Square it and change to polar coordinates:
\begin{align}
  I^2 &= \int_{-\infty}^{\infty}\!\int_{-\infty}^{\infty} e^{-(x^2+y^2)}\,dx\,dy \\
      &= \int_{0}^{2\pi}\!\int_{0}^{\infty} e^{-r^2}\, r\,dr\,d\theta
       = 2\pi \left[-\tfrac12 e^{-r^2}\right]_{0}^{\infty} = \pi,
\end{align}
so $\displaystyle \int_{-\infty}^{\infty} e^{-x^2}\,dx = \sqrt{\pi}$.

\section*{Cases and sums}
\begin{equation}
  f(x) =
  \begin{cases}
    x^2 \sin\frac{1}{x}, & x \neq 0,\\[2pt]
    0,                   & x = 0,
  \end{cases}
  \qquad
  \sum_{n=1}^{\infty} \frac{1}{n^2} = \frac{\pi^2}{6},
  \qquad
  \prod_{p\ \text{prime}} \frac{1}{1-p^{-s}} = \zeta(s).
\end{equation}

\section*{Matrices}
\begin{equation}
  \det\begin{pmatrix} a & b & c\\ d & e & f\\ g & h & i \end{pmatrix}
  = a\begin{vmatrix} e & f\\ h & i \end{vmatrix}
  - b\begin{vmatrix} d & f\\ g & i \end{vmatrix}
  + c\begin{vmatrix} d & e\\ g & h \end{vmatrix},
  \qquad
  e^{tA} = \sum_{k\ge 0} \frac{t^k A^k}{k!},\;
  A = \begin{bmatrix} 0 & -1\\ 1 & 0 \end{bmatrix}.
\end{equation}

\section*{A continued fraction}
\begin{equation}
  \frac{4}{\pi} = 1 + \cfrac{1^2}{2 + \cfrac{3^2}{2 + \cfrac{5^2}{2 + \cfrac{7^2}{2 + \dotsb}}}}
\end{equation}
