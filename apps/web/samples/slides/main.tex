\begin{frame}{Why it feels instant}
  \begin{columns}[T]
    \begin{column}{.48\textwidth}
      \begin{itemize}
        \item<1-> A deck is usually rebuilt from scratch.
        \item<2-> PhiTeX rebuilds only the frame you edit.
        \item<3-> The others are kept, page for page.
      \end{itemize}
    \end{column}
    \begin{column}{.48\textwidth}
      \begin{block}{Exact}
        The PDF is byte for byte the one a full \texttt{pdflatex} run makes.
      \end{block}
      \begin{alertblock}{Fast}
        A keystroke costs a frame, not a deck.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
