\begin{frame}{Why incremental?}
  \begin{itemize}
    \item A slide deck is rebuilt from scratch on every compile.
    \item PhiTeX rebuilds only the frame you are editing.
    \item The other frames are kept as they were.
  \end{itemize}
\end{frame}

\begin{frame}{A formula}
  \[
    e^{i\pi} + 1 = 0
  \]

\end{frame}
